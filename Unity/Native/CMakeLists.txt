cmake_minimum_required(VERSION 3.16)
project(FitdUnityGpuBridge LANGUAGES CXX)
add_library(FitdUnityGpuBridge STATIC unityGpuBridge.cpp)
target_compile_features(FitdUnityGpuBridge PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(FitdUnityGpuBridge PRIVATE /Zc:__cplusplus /Zc:preprocessor)
endif()
target_include_directories(FitdUnityGpuBridge PRIVATE ../../ThirdParty/bgfx.cmake/bgfx/include ../../ThirdParty/bgfx.cmake/bx/include)
find_package(Threads REQUIRED)
add_executable(FitdUnityGpuSmoke ../Tests/gpuBridgeSmoke.cpp)
target_compile_features(FitdUnityGpuSmoke PRIVATE cxx_std_17)
target_link_libraries(FitdUnityGpuSmoke PRIVATE FitdUnityGpuBridge Threads::Threads)
target_include_directories(FitdUnityGpuSmoke PRIVATE ../../ThirdParty/bgfx.cmake/bgfx/include ../../ThirdParty/bgfx.cmake/bx/include)
if(MSVC)
    target_compile_options(FitdUnityGpuSmoke PRIVATE /Zc:__cplusplus /Zc:preprocessor)
endif()
enable_testing()
add_test(NAME unityGpuBridge COMMAND FitdUnityGpuSmoke)
