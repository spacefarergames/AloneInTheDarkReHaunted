option(FITD_UNITY_COMMANDS "Render through Unity's GPU command buffers and Unity audio" ON)
option(FITD_UNITY_BUNDLED_DATA_ONLY "Ship gameplay data in Unity bundles instead of compiled arrays" ON)
if((NOT WIN32 OR UWP) AND NOT FITD_UNITY_COMMANDS)
    message(FATAL_ERROR "The Unity host currently supports Windows x64 desktop only")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The Unity gameplay host requires a 64-bit target")
endif()

# Compile a separate engine variant so the standalone executable stays independent.
get_target_property(ENGINE_SOURCES FitdLib SOURCES)
get_target_property(ENGINE_INCLUDES FitdLib INCLUDE_DIRECTORIES)
set(UNITY_ENGINE_SOURCES)
foreach(SOURCE IN LISTS ENGINE_SOURCES)
    if(FITD_UNITY_COMMANDS AND FITD_UNITY_BUNDLED_DATA_ONLY AND SOURCE MATCHES "[/\\\\]embedded[/\\\\]")
        continue()
    endif()
    if(SOURCE MATCHES "[/\\\\]LockedFiles[/\\\\]" OR SOURCE MATCHES "_temp\\.cpp$")
        continue()
    endif()
    if(NOT IS_ABSOLUTE "${SOURCE}")
        set(SOURCE "${PROJECT_SOURCE_DIR}/FitdLib/${SOURCE}")
    endif()
    list(APPEND UNITY_ENGINE_SOURCES "${SOURCE}")
endforeach()
add_library(FitdUnity SHARED ${UNITY_ENGINE_SOURCES} Native/unityHost.cpp)
target_include_directories(FitdUnity PRIVATE ${ENGINE_INCLUDES} "${PROJECT_SOURCE_DIR}/FitdLib")
target_compile_features(FitdUnity PRIVATE cxx_std_17)
target_compile_definitions(FitdUnity PRIVATE FITD_UNITY=1)
if(MSVC)
    target_compile_options(FitdUnity PRIVATE /Zc:__cplusplus /Zc:preprocessor /EHsc)
endif()
if(FITD_UNITY_COMMANDS)
    target_compile_definitions(soloud PRIVATE WITH_NULL=1)
    target_sources(FitdUnity PRIVATE Native/unityGpuBridge.cpp Native/unityFileSystem.cpp Native/unitySettings.cpp)
    if(FITD_UNITY_BUNDLED_DATA_ONLY)
        target_sources(FitdUnity PRIVATE Native/unityBundledData.cpp)
    endif()
    target_compile_definitions(FitdUnity PRIVATE FITD_UNITY_COMMANDS=1 STBI_WINDOWS_UTF8=1 STBIW_WINDOWS_UTF8=1)
    target_link_libraries(FitdUnity PRIVATE bx bimg zlibstatic SDL3-static soloud)
else()
    target_link_libraries(FitdUnity PRIVATE bgfx zlibstatic SDL3-static soloud)
endif()
set_target_properties(FitdUnity PROPERTIES OUTPUT_NAME FitdUnity)
if(FITD_UNITY_COMMANDS)
    set_target_properties(FitdUnity PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
    if(UNIX AND NOT APPLE)
        target_link_options(FitdUnity PRIVATE "-Wl,--exclude-libs,ALL")
    endif()
endif()
